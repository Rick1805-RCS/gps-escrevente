function doPost(e) { try { var data = JSON.parse(e.postData.contents); // 1. Se for solicitação de carregamento do GPS Principal if (data.action === "load") { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Dados"); var cell = sheet.getRange("A1").getValue(); return ContentService.createTextOutput(JSON.stringify({"status": "success", "data": JSON.parse(cell)})) .setMimeType(ContentService.MimeType.JSON); } // 2. Se for o envio silencioso de redação do novo módulo separado if (data.action === "redacao") { var emailDestino = data.email; var dados = data.dados; if (emailDestino && emailDestino.indexOf("@") !== -1 && dados && dados.redacao) { var assunto = "📝 Nova Redação Enviada - " + (dados.tema || "Treino VUNESP"); var corpo = "Uma nova redação foi gerada/salva pela aluna!\n\n" + "Tema: " + dados.tema + "\n\n" + "=== 1. INTRODUÇÃO ===\n" + dados.redacao.intro + "\n\n" + "=== 2. DESENVOLVIMENTO 1 ===\n" + dados.redacao.dev1 + "\n\n" + "=== 3. DESENVOLVIMENTO 2 ===\n" + dados.redacao.dev2 + "\n\n" + "=== 4. CONCLUSÃO ===\n" + dados.redacao.concl; MailApp.sendEmail(emailDestino, assunto, corpo); } return ContentService.createTextOutput(JSON.stringify({"status": "success"})) .setMimeType(ContentService.MimeType.JSON); } // 3. Comportamento padrão para salvar o seu GPS principal if (data.email && data.dados) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Dados"); sheet.getRange("A1").setValue(JSON.stringify(data.dados)); return ContentService.createTextOutput(JSON.stringify({"status": "success"})) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({"status": "success"})) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": error.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
